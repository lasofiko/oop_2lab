cmake_minimum_required(VERSION 3.16)
project(oop_2lab LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
if(ENABLE_ASAN)
    if(MSVC)
        add_compile_options(/fsanitize=address)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address)
    else()
        message(FATAL_ERROR "AddressSanitizer is unsupported for this compiler")
    endif()
endif()

add_library(polynomial src/polynomial.cpp)
target_include_directories(polynomial PUBLIC include)
add_executable(polynomial_demo src/main.cpp)
target_link_libraries(polynomial_demo PRIVATE polynomial)

include(CTest)
if(BUILD_TESTING)
    find_package(GTest QUIET)
    if(NOT GTest_FOUND)
        include(FetchContent)
        set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
        FetchContent_Declare(googletest
            GIT_REPOSITORY https://github.com/google/googletest.git
            GIT_TAG v1.14.0
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(polynomial_tests tests/test_polynomial.cpp)
    if(TARGET GTest::gtest_main)
        target_link_libraries(polynomial_tests PRIVATE polynomial GTest::gtest_main)
    else()
        target_link_libraries(polynomial_tests PRIVATE polynomial GTest::Main)
    endif()
    include(GoogleTest)
    gtest_discover_tests(polynomial_tests)
endif()
